# build image from Dockerfile
docker build -t <name>:<tag> .

# create container from image with bash entrypoint
docker run -it --rm --entrypoint=bash <name>:<tag>

# run postgres within a container
docker run -it --rm \
	-e POSTGRES_USER="root" \
	-e POSTGRES_PASSWORD="root" \
	-e POSTGRES_DB="ny_taxi" \
	-v ny_taxi_postgres_data:/var/lib/postgresql \
	-p 5431:5432 \
	postgres:18

# run pipeline locally
uv run ingest_data.py \
 --pg-user=root \
 --pg-pass=root \
 --pg-host=localhost \
 --pg-db=ny_taxi \
 --pg-port=5431 \
 --year=2019 \
 --month=10 \
 --chunk-size=100_000 \
 --target-table=yellow_taxi_data \
 --replace=True


# DOCKERIZED PIPELINE
# create docker image
docker build -t taxi_ingest:v001 .

# create new network
docker network create pg-network

# run postgres in the network
docker run -it \
  -e POSTGRES_USER="root" \
  -e POSTGRES_PASSWORD="root" \
  -e POSTGRES_DB="ny_taxi" \
  -v ny_taxi_postgres_data:/var/lib/postgresql \
  -p 5431:5432 \
  --network=pg-network \
  --name=pgdatabase \
  postgres:18

# run pgadmin in the network
docker run -it \
	-e PGADMIN_DEFAULT_EMAIL="admin@admin.com" \
  -e PGADMIN_DEFAULT_PASSWORD="root" \
  -v pgadmin_data:/var/lib/pgadmin \
  -p 8085:80 \
  --network=pg-network \
  --name=pgadmin \
  dpage/pgadmin4

# run the pipeline in the network
docker run -it --rm --network=pg-network taxi_ingest:v001 \
 --pg-user=root \
 --pg-pass=root \
 --pg-host=pgdatabase \
 --pg-db=ny_taxi \
 --pg-port=5432 \
 --year=2019 \
 --month=10 \
 --chunk-size=100_000 \
 --target-table=yellow_taxi_data \
 --replace=True
